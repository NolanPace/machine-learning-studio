cmake_minimum_required(VERSION 3.16)
project(MLStudio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(linear STATIC linear.cpp)
target_include_directories(linear PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(ml_studio main.cpp)
target_link_libraries(ml_studio PRIVATE linear)
enable_testing()
add_executable(ml_tests tests.cpp)
target_link_libraries(ml_tests PRIVATE linear)
add_test(NAME MLTests COMMAND ml_tests)
